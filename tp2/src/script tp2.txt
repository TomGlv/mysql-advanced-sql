CREATE DATABASE IF NOT EXISTS tp2_mysql;
USE tp2_mysql;

-- Suppression des anciennes tables pour pouvoir
-- réinitialiser le TP. Attention : cela efface leurs données.

DROP TABLE IF EXISTS affectations_projets;
DROP TABLE IF EXISTS projets;
DROP TABLE IF EXISTS employes;
DROP TABLE IF EXISTS clients;
DROP TABLE IF EXISTS departements;

-- ==================================================
-- 1. CREATION DES TABLES
-- ==================================================

CREATE TABLE departements (
    id INT PRIMARY KEY AUTO_INCREMENT,
    nom VARCHAR(100) NOT NULL
);

CREATE TABLE employes (
    id INT PRIMARY KEY AUTO_INCREMENT,
    prenom VARCHAR(50) NOT NULL,
    nom VARCHAR(50) NOT NULL,
    salaire DECIMAL(10,2) NOT NULL,
    departement_id INT,
    FOREIGN KEY (departement_id)
        REFERENCES departements(id)
);

CREATE TABLE clients (
    id INT PRIMARY KEY AUTO_INCREMENT,
    nom VARCHAR(100) NOT NULL
);

CREATE TABLE projets (
    id INT PRIMARY KEY AUTO_INCREMENT,
    nom VARCHAR(100) NOT NULL,
    budget DECIMAL(12,2) NOT NULL,
    client_id INT,
    FOREIGN KEY (client_id)
        REFERENCES clients(id)
);

-- Table d'association entre employés et projets
CREATE TABLE affectations_projets (
    id INT PRIMARY KEY AUTO_INCREMENT,
    employe_id INT NOT NULL,
    projet_id INT NOT NULL,
    role VARCHAR(100),
    FOREIGN KEY (employe_id)
        REFERENCES employes(id),
    FOREIGN KEY (projet_id)
        REFERENCES projets(id),
    UNIQUE (employe_id, projet_id)
);
